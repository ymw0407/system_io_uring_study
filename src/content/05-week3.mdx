## 이번 실습의 목표

앞 절에서 CPU 사용률과 QD 라는 두 신호, hysteresis 와 cooldown 으로 정책의 뼈대를 정리했다. 이번 실습에서는 그 뼈대를 *직접 C 로* 구현하고 NVMe 위에서 실제로 작동시켜 본다.

## 왜 C로 짜야 하는가

표준 fio 의 ioengine 은 interrupt · SQPOLL · IOPOLL 같은 고정 모드만 지원한다 — CPU 와 QD 를 보고 런타임에 모드를 전환하는 동적 정책은 들어 있지 않다. 정책의 메커니즘을 가장 직접적으로 이해하는 길은 사용자-공간 C 코드로 한 번 만들어 보는 것이다. 구현 후에는 fio 로 측정한 고정 모드 기준선들과 비교하여 adaptive 의 효과를 검증한다.

## 정책 설계

정책은 두 가지 신호를 관찰한다. 첫 번째는 `/proc/stat`에서 읽는 CPU 사용률이다. 70%를 넘으면 SQPOLL 스레드가 다른 작업과 CPU를 두고 경쟁하므로 interrupt로 전환한다. 두 번째는 현재 in-flight 요청 수(QD)로, 8 이상이면서 CPU에 여유가 있으면 polling으로 전환한다. 모드 전환 자체에 비용이 있으므로 500밀리초 쿨다운을 두어 진동을 방지한다.

<CodeBlock language="plaintext" filename="adaptive_policy.txt">
{`every 200ms:
    cpu_usage = read_cpu_usage_from_proc_stat()
    current_qd = count_inflight_requests()

    if time_since_last_switch < 500ms:
        return   // cooldown: do not switch

    if cpu_usage > 70%:
        switch_to_interrupt_mode()
    else if current_qd >= 8 and cpu_usage <= 70%:
        switch_to_polling_mode()
    // otherwise: keep current mode`}
</CodeBlock>

## 구현하기

<Steps>
  <Step title="CPU 사용률 읽기 함수">

`/proc/stat`의 첫 줄은 시스템 전체 CPU 시간을 나타낸다. 두 시점의 값을 비교하면 그 구간의 CPU 사용률을 계산할 수 있다.

<CodeBlock language="c" filename="cpu_usage.c">
{`struct cpu_sample {
    unsigned long long user, nice, system, idle;
    unsigned long long iowait, irq, softirq, steal;
};

static int read_cpu(struct cpu_sample *s) {
    FILE *f = fopen("/proc/stat", "r");
    if (!f) return -1;
    fscanf(f, "cpu %llu %llu %llu %llu %llu %llu %llu %llu",
           &s->user, &s->nice, &s->system, &s->idle,
           &s->iowait, &s->irq, &s->softirq, &s->steal);
    fclose(f);
    return 0;
}

static double calc_cpu_usage(struct cpu_sample *prev,
                             struct cpu_sample *cur) {
    unsigned long long prev_idle = prev->idle + prev->iowait;
    unsigned long long cur_idle  = cur->idle  + cur->iowait;
    unsigned long long prev_total =
        prev->user + prev->nice + prev->system +
        prev->idle + prev->iowait + prev->irq +
        prev->softirq + prev->steal;
    unsigned long long cur_total =
        cur->user + cur->nice + cur->system +
        cur->idle + cur->iowait + cur->irq +
        cur->softirq + cur->steal;
    unsigned long long total_d = cur_total - prev_total;
    unsigned long long idle_d  = cur_idle  - prev_idle;
    if (total_d == 0) return 0.0;
    return 100.0 * (1.0 - (double)idle_d / (double)total_d);
}`}
</CodeBlock>

  <Step title="두 링 초기화">

Adaptive 전환의 핵심은 interrupt 모드 링과 SQPOLL 모드 링을 프로그램 시작 시 동시에 만들어두는 것이다. 모드를 바꿀 때 링을 파괴하고 재생성하면 비용이 크므로, 활성 링 포인터만 교체하는 방식을 사용한다.

<CodeBlock language="c" filename="init_two_rings.c">
{`/* interrupt mode ring */
struct io_uring ring_int;
if (io_uring_queue_init(QD, &ring_int, 0) < 0) {
    perror("io_uring_queue_init (interrupt)");
    return 1;
}

/* polling mode ring (SQPOLL) */
struct io_uring ring_poll;
struct io_uring_params p = {0};
p.flags = IORING_SETUP_SQPOLL;
p.sq_thread_idle = 2000;
if (io_uring_queue_init_params(QD, &ring_poll, &p) < 0) {
    perror("io_uring_queue_init_params (polling)");
    io_uring_queue_exit(&ring_int);
    return 1;
}

enum io_mode { MODE_INTERRUPT, MODE_POLLING };
enum io_mode current = MODE_INTERRUPT;
struct io_uring *active = &ring_int;`}

  <Step title="전체 코드 작성">

앞서 작성한 CPU 읽기 함수와 두 링 초기화를 조합하여 전체 벤치마크 프로그램을 완성한다.

<CodeBlock language="c" filename="bench_adaptive.c">
{`#define _GNU_SOURCE
#include <liburing.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

#define QD          32
#define BS          4096
#define CPU_THR     70.0
#define QD_THR      8
#define COOLDOWN_MS 500
#define RUNTIME_SEC 10
#define CHECK_MS    200   /* jiffy resolution 노이즈 방지 */

struct cpu_sample {
    unsigned long long user, nice, system, idle;
    unsigned long long iowait, irq, softirq, steal;
};

static int read_cpu(struct cpu_sample *s) {
    FILE *f = fopen("/proc/stat", "r");
    if (!f) return -1;
    int n = fscanf(f, "cpu %llu %llu %llu %llu %llu %llu %llu %llu",
                   &s->user, &s->nice, &s->system, &s->idle,
                   &s->iowait, &s->irq, &s->softirq, &s->steal);
    fclose(f);
    return (n == 8) ? 0 : -1;
}

static long ms_diff(struct timespec *a, struct timespec *b) {
    return (b->tv_sec - a->tv_sec) * 1000L
         + (b->tv_nsec - a->tv_nsec) / 1000000L;
}

static double calc_cpu_usage(struct cpu_sample *prev,
                             struct cpu_sample *cur) {
    unsigned long long prev_idle = prev->idle + prev->iowait;
    unsigned long long cur_idle  = cur->idle  + cur->iowait;
    unsigned long long prev_total =
        prev->user + prev->nice + prev->system +
        prev->idle + prev->iowait + prev->irq +
        prev->softirq + prev->steal;
    unsigned long long cur_total =
        cur->user + cur->nice + cur->system +
        cur->idle + cur->iowait + cur->irq +
        cur->softirq + cur->steal;
    unsigned long long total_d = cur_total - prev_total;
    unsigned long long idle_d  = cur_idle  - prev_idle;
    if (total_d == 0) return 0.0;
    return 100.0 * (1.0 - (double)idle_d / (double)total_d);
}

enum io_mode { MODE_INTERRUPT, MODE_POLLING };

int main(int argc, char *argv[]) {
    if (argc < 2) {
        fprintf(stderr, "usage: %s <file>\n", argv[0]);
        return 1;
    }

    int fd = open(argv[1], O_RDONLY | O_DIRECT);
    if (fd < 0) { perror("open"); return 1; }

    /* Initialize two rings: interrupt and polling */
    struct io_uring ring_int, ring_poll;
    if (io_uring_queue_init(QD, &ring_int, 0) < 0) {
        perror("io_uring_queue_init (interrupt)");
        return 1;
    }

    struct io_uring_params p = {0};
    p.flags = IORING_SETUP_SQPOLL;
    p.sq_thread_idle = 2000;
    if (io_uring_queue_init_params(QD, &ring_poll, &p) < 0) {
        perror("io_uring_queue_init_params (polling)");
        io_uring_queue_exit(&ring_int);
        return 1;
    }

    /* fd 를 두 ring 모두에 register. SQPOLL 의 옛 커널 호환 + 가상화
     * 환경 안정성. prep_read 가 fd 대신 fixed-file index(0) 를 받는다. */
    int reg_fds[] = { fd };
    if (io_uring_register_files(&ring_int, reg_fds, 1) < 0) {
        perror("io_uring_register_files (interrupt)");
        return 1;
    }
    if (io_uring_register_files(&ring_poll, reg_fds, 1) < 0) {
        perror("io_uring_register_files (polling)");
        return 1;
    }

    enum io_mode current = MODE_INTERRUPT;
    struct io_uring *active = &ring_int;

    struct cpu_sample prev_cpu, cur_cpu;
    read_cpu(&prev_cpu);

    struct timespec last_switch;
    clock_gettime(CLOCK_MONOTONIC, &last_switch);

    void *buf;
    if (posix_memalign(&buf, BS, BS) != 0) {
        perror("posix_memalign");
        return 1;
    }

    int inflight = 0;

    struct timespec run_start;
    clock_gettime(CLOCK_MONOTONIC, &run_start);
    struct timespec last_check = run_start;

    int i = 0;
    while (1) {
        /* Submit a read request */
        struct io_uring_sqe *sqe = io_uring_get_sqe(active);
        if (!sqe) {
            /* SQ 가 꽉 찼다. SQPOLL 에서는 sqring_wait, 일반 모드에서는
             * wait_cqe 로 슬롯을 회수한다. */
            if (current == MODE_POLLING) {
                if (io_uring_sqring_wait(active) < 0) break;
            } else {
                struct io_uring_cqe *cqe;
                if (io_uring_wait_cqe(active, &cqe) < 0) break;
                io_uring_cqe_seen(active, cqe);
                inflight--;
            }
            sqe = io_uring_get_sqe(active);
            if (!sqe) break;
        }
        /* registered file index (0) + IOSQE_FIXED_FILE. raw fd 안 씀. */
        io_uring_prep_read(sqe, 0, buf, BS, 0);
        sqe->flags |= IOSQE_FIXED_FILE;
        if (io_uring_submit(active) < 0) break;
        inflight++;

        /* Reap completions */
        struct io_uring_cqe *cqe;
        while (io_uring_peek_cqe(active, &cqe) == 0) {
            io_uring_cqe_seen(active, cqe);
            inflight--;
        }

        /* 정책 체크 — 측정 윈도우가 jiffy 보다 짧으면 cpu_pct 가 0%/100%
         * 사이에서 양자화 노이즈로 진동한다. CHECK_MS 만큼 시간이 흘렀을
         * 때만 실제 CPU 측정과 정책 결정을 한다. */
        if (i % 1000 == 0 && i > 0) {
            struct timespec now;
            clock_gettime(CLOCK_MONOTONIC, &now);

            if (ms_diff(&run_start, &now) >= RUNTIME_SEC * 1000) break;
            if (ms_diff(&last_check, &now) < CHECK_MS) continue;

            read_cpu(&cur_cpu);
            double cpu = calc_cpu_usage(&prev_cpu, &cur_cpu);
            prev_cpu = cur_cpu;
            last_check = now;

            if (ms_diff(&last_switch, &now) < COOLDOWN_MS) continue;

            if (cpu > CPU_THR && current == MODE_POLLING) {
                current = MODE_INTERRUPT;
                active = &ring_int;
                last_switch = now;
                printf("[switch] -> INTERRUPT (cpu=%.1f%%)\n", cpu);
            } else if (inflight >= QD_THR && cpu <= CPU_THR
                       && current == MODE_INTERRUPT) {
                current = MODE_POLLING;
                active = &ring_poll;
                last_switch = now;
                printf("[switch] -> POLLING (cpu=%.1f%%, qd=%d)\n",
                       cpu, inflight);
            }
        }

        i++;
    }

    printf("Done after %d iterations. Final mode: %s\n", i,
           current == MODE_POLLING ? "POLLING" : "INTERRUPT");

    io_uring_queue_exit(&ring_int);
    io_uring_queue_exit(&ring_poll);
    free(buf);
    close(fd);
    return 0;
}`}
</CodeBlock>

<Callout kind="note">
`io_uring_submit()`은 interrupt 모드 링에서는 `io_uring_enter` 시스템 콜을 발생시키지만, SQPOLL 링에서는 커널 스레드가 슬립 중일 때만 깨우는 역할을 한다. 따라서 두 링 모두에 동일한 호출을 사용해도 동작이 올바르다.
</Callout>

  </Step>

</Callout>

## fio baseline과 비교하기

<Callout kind="warn">
이 비교에는 구조적 한계가 있다. adaptive 결과는 직접 짠 C 벤치마크로, interrupt와 SQPOLL 결과는 fio로 측정했다. 측정 harness가 다르므로 절대값의 직접 비교는 부정확하다. 동일 디바이스(NVMe), 동일 블록 크기(4KB), 동일 런타임을 맞추어 최대한 공정하게 비교했지만, harness 차이에서 오는 오차를 감안해야 한다.
</Callout>

fio로 interrupt와 SQPOLL 기준선을 먼저 측정하고, 이어서 adaptive C 벤치마크를 실행한다.

이 NVMe 에서는 세 모드의 지연이 *거의 같다*. QD=16 기준 fio interrupt 48.9μs, fio SQPOLL 48.5μs, adaptive C 50.1μs 로 약 1~3% 차이. SN950X 의 디바이스 시간이 latency 의 95%+ 를 차지해서 OS path 비용이 묻히는 환경 — Part 3 에서 본 *polling 의 latency 이득 영역은 좁다* 라는 관찰이 베어메탈 데이터로 다시 확인된다.

adaptive C 가 QD=1, 4 에서 fio interrupt 와 *정확히 일치*하는 것은 정책의 명백한 신호다. 이 영역에서는 `inflight < qd_lo=8` 이라 정책이 POLLING 으로 전환하지 않고 INTERRUPT 를 유지한다. QD=16 부터는 POLLING 으로 옮기지만 SQPOLL 자체의 이득이 작아 latency 차이도 작다.

여기서 adaptive 의 진짜 가치가 드러난다. Idle / 50% 부하에서 adaptive 는 1244 MB/s 로 fio SQPOLL 의 1284 와 fio interrupt 의 1268 사이에 위치한다 (정책이 시간의 ~94% 를 SQPOLL 로 보낸 결과). 그런데 *CPU 100% 부하* 에서:

- fio SQPOLL 은 1284 → 531 MB/s 로 *58% 처리량 손실* — SQPOLL kthread 가 자기 코어를 stress-ng 와 경쟁하면서 polling 의 가정이 깨진 모습.
- fio interrupt 는 1268 → 983 MB/s, 22% 손실로 견딘다.
- **adaptive 는 1044 MB/s 로 두 모드 모두 이긴다.** 정책이 CPU 100% 를 인지하고 INTERRUPT 로 전환한 결과 (polling fraction 이 idle 94% 에서 100% 부하에선 0% 로 떨어짐).

<Callout kind="checkpoint">
세 가지가 다 맞아야 정책이 동작하는 것이다.
1. Idle 처리량이 fio SQPOLL 에 근접 (1244 ≈ 1284)
2. CPU 100% 처리량이 fio SQPOLL 의 cliff (531) 를 *회피*하고 fio interrupt 이상 (1044 ≥ 983)
3. Polling fraction 이 부하에 따라 갈림 (Idle 94% → CPU 100% 부하 0%)
</Callout>

두 그래프의 라인/바 사이 보간은 실제 측정 사이의 동작을 의미하지 않는다. 각 데이터 포인트는 해당 조건에서의 개별 측정값이다.
